# lanzou-direct-link —— 依赖 / dependencies
#
# === 核心库 / core library（必需 / required）===
# 纯 requests：解析编排、过 CDN 反爬、带进度/可取消的下载
requests>=2.25

# === 图形界面（可选）/ GUI (optional) ===
# 只有 lanzou_gui.py 需要 PyQt6：
#     pip install PyQt6
PyQt6>=6.5

# === 浏览器引擎（解析必需）/ browser engine (required for parsing) ===
# 现代蓝奏云的分享页是 JS 反爬挑战页，纯 HTTP 拿不到直链，必须真实浏览器渲染。
# 装完还要执行一次：
#     playwright install chromium
# （目标机器已有 Edge/Chrome 时那一步可以跳过 —— 代码会依次尝试
#   channel="msedge" → channel="chrome" → 自带的 chromium）
playwright>=1.30
